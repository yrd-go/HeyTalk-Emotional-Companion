#     pip install -r requirements.txt

# ---------------------------------------------------------------------------
# MCP 模块依赖（mcp_weather_server.py / mcp_weather_client.py /
#               mcp_protocol_probe.py 使用）
#
# 安装：
#     pip install -r requirements.txt
#
# 说明：
#   - mcp 是 MCP（Model Context Protocol）官方 Python SDK，提供 MCPServer
#     （mcp 1.x 里叫 FastMCP）与服务端 / 客户端实现；服务端走 stdio 传输，
#     由 LangChain 客户端以子进程方式拉起，不占用端口。
#     [cli] extra 会额外装上命令行工具（mcp dev / mcp run / mcp install），
#     其中 mcp dev（MCP Inspector）需要本机已安装 Node/npx。
#   - langchain[mcp] 与上面 RAG 段的 langchain 是同一个发行包，这里通过
#     extra 声明 MCP 适配层：MCP 能力已由独立的 langchain-mcp-adapters
#     迁入 langchain.mcp 命名空间（MCPAdapter），该命名空间要求
#     langchain>=1.4.0，目前为 beta，导入时会提示 LangChainBetaWarning。
#   - 两者都要求 Python >= 3.10；若解释器为 3.14，仍按上面 RAG 段关于
#     chromadb 的注意事项处理。
#   - mcp_weather_client.py 在 langchain.mcp 不可用时会自动回退到旧的
#     langchain-mcp-adapters（已停止积极维护），因此本文件不强制安装它，
#     见下方注释行。
# ---------------------------------------------------------------------------

# MCP 协议官方 Python SDK + 命令行工具（Server / Client / 协议探针共用）
# 锁 >=2.2 的原因（踩过的坑）：2.2 起 stdio 服务端会"抢占" fd 1 并把 fd 1 指向 stderr
# （mcp/server/stdio.py，上游 issue #1933），从工具内部打印再也污染不到协议线。
# 这直接改变了 src/mcp_demo 负向样例的行为：旧版 SDK 下能复现协议层失败（退出码 4），
# 新版下复现不出来（探针如实返回退出码 5）。不锁版本会导致 CI 与本地装到不同 SDK、
# 出现"CI 绿而本地红"的假绿 —— 所以这里显式锁定范围。
mcp[cli]>=2.2,<3

# LangChain 内置 MCP 适配层（MCPAdapter，需要 langchain>=1.4.0）
langchain[mcp]>=1.4.0

# 可选回退：旧版适配层，仅当 langchain.mcp 不可用时才需要手动安装
# （mcp_weather_client.py 会自动探测并回退到它）
# langchain-mcp-adapters
